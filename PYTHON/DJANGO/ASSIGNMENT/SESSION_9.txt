from django.core.mail import EmailMultiAlternatives
from django.template.loader import render_to_string

def place_order(request):
    html_message = render_to_string(
        "order_confirmation.html",
        {
            "username": "Hiren",
            "order_id": 101,
            "amount": 499,
        }
    )

    email = EmailMultiAlternatives(
        subject="Order Confirmation",
        body="Your order has been placed successfully.",
        from_email="yourgmail@gmail.com",
        to=["customer@gmail.com"])

    email.attach_alternative(html_message, "text/html")
    email.send()
    return HttpResponse("Order Confirmation Email Sent")

from django.http import HttpResponse
from django.core.mail import EmailMultiAlternatives
from django.template.loader import render_to_string

def order_confirmation(request):

    context = {
        "username": "Hiren",
        "order_id": 101,
        "food_name": "Paneer Pizza",
        "quantity": 2,
        "amount": 499
    }

    html_message = render_to_string("order_confirmation.html", context)

    email = EmailMultiAlternatives(
        subject="Your Order is Confirmed!",
        body="Order placed successfully.",
        from_email="yourgmail@gmail.com",
        to=["yourgmail@gmail.com"]

    )

    email.attach_alternative(
        html_message,
        "text/html"
    )

    email.send()

    return HttpResponse("Sample Order Confirmation Email Sent")

from django.http import HttpResponse
from django.core.mail import EmailMultiAlternatives
from django.template.loader import render_to_string
def send_welcome_email(request):
    context = {
        "username": "Hiren",
        "team_name": "Super Strikers",
    }
    html_message = render_to_string("welcome_email.html",context)
    subject = "Welcome to IPL Fantasy League Build Your Dream Team!"
    body = """Welcome to IPL Fantasy League!
        Your account has been created successfully.
        Start building your dream team, join contests,
        earn points, and compete with cricket fans.
        Best of luck!
        IPL Fantasy League Team"""

    email = EmailMultiAlternatives(subject=subject,body=body,from_email="yourgmail@gmail.com",to=["yourgmail@gmail.com"])
