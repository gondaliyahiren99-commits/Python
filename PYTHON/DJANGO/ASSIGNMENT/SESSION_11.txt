2.Create a Django model called InfluencerProfile with fields: user (OneToOneField), display_name (CharField), bio (TextField), and profile_pic (ImageField). Ensure profile_pic uploads to 'profile_pics/'.
from django.db import models
from django.contrib.auth.models import User

class InfluencerProfile(models.Model):
    user = models.OneToOneField( User,on_delete=models.CASCADE)
    display_name = models.CharField(max_length=100)
    bio = models.TextField()
    profile_pic = models.ImageField(upload_to="profile_pics/")

    def __str__(self):
        return self.display_name

3.Build a profile edit page where users can update their display name, bio, and upload a new profile picture, similar to editing your Instagram profile.
from django.shortcuts import render, redirect
from .models import InfluencerProfile
from .forms import InfluencerProfileForm

def edit_profile(request):
    profile = InfluencerProfile.objects.get(user=request.user)
    if request.method == "POST":
        form = InfluencerProfileForm(
            request.POST,
            request.FILES,
            instance=profile
        )
        if form.is_valid():
            form.save()
            return redirect("edit_profile")
    else:
        form = InfluencerProfileForm(instance=profile)

    return render(request,"edit_profile.html",{"form": form})
