Q4.Add search functionality in the Restaurant admin so you can search restaurants by name or cuisine using the search_fields option.
 ~~>  
    from django.contrib import admin
    from .models import Restaurant

    class RestaurantAdmin(admin.ModelAdmin):
        list_display = ('name', 'cuisine', 'rating')
        search_fields = ('name', 'cuisine')

    admin.site.register(Restaurant, RestaurantAdmin)

Q5.Add a filter in the Restaurant admin panel to filter restaurants by cuisine using the list_filter option.Use list_filter = ['cuisine'] in your RestaurantAdmin class
 ~~>
    from django.contrib import admin
    from .models import Restaurant

    class RestaurantAdmin(admin.ModelAdmin):
        list_display = ('name', 'cuisine', 'rating')
        search_fields = ('name', 'cuisine')
        list_filter = ['cuisine']

    admin.site.register(Restaurant, RestaurantAdmin)
