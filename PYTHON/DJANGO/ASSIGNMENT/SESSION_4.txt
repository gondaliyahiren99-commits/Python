class Restaurant(models.Model):
    name=models.CharField(max_length = 30,unique=True)
    location=models.CharField(max_length = 30)
    rating=models.FloatField()
    cuisine = models.ForeignKey(Cuisine,on_delete=models.CASCADE)

    def __str__(self):
        return self.name
