Q4.4. File handling
Build a program that:
● Writes daily goals to a file
● Reads and displays them on program restart
● Uses correct file modes
#include<stdio.h>
void add()
    {   
        FILE *p;
        char str[100];
        printf("Enter Goal :");
        fgets(str,sizeof(str),stdin);
        p=fopen("Daily_Goal,txt","a");
        fprintf(p,"%s",str);
        fclose(p);
        
    }
void view()
    {
        char data[100];
        FILE *p;
        p=fopen("Daily_Goal.txt","r");
        if(p==NULL)
            {
                printf("Not found");
            }
        while(fgets(data,100,p)!=NULL)
            {
                printf("%s",data);
            }
        fclose(p);
    }
int main()
    {
        while(1)
            {
                int choice;
                printf("\n---------Menu---------\n");
                printf("1.Add Goal\n");
                printf("2.View Goal\n");
                printf("3.Remove Goal\n");
                printf("4.Exit\n");
                printf("Enter choice :");
                scanf("%d",&choice);
                if(choice==1)
                    {
                        add();
                    }
                else if(choice==2)
                    {
                        view();
                    }
                else if(choice==4)
                    {
                        break;
                    }
                else{
                    printf("Invalid");
                }
            }
    }
